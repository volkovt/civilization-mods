-- StartRepositionCycle.lua (BNW-friendly / Universal)
--
-- Behavior (as requested):
-- 1) Prev/Next swaps IMMEDIATELY (no dependency on Confirm).
-- 2) Confirm only hides the panel (finalize).
-- 3) Panel is visible on Turn 0 only, and auto-hides when:
--      - the human player manually moves any unit, OR
--      - the turn advances.
-- 4) Manual coords ALWAYS attempts to move you to the requested location:
--      - If the destination plot is occupied by another civ's unit(s), we swap:
--          other civ -> your previous center
--          you       -> requested destination
--      - If the exact plot is invalid (water/mountain/impassable/city), we choose the
--        nearest valid land plot within a search radius, but we still treat it as the
--        "requested location" intent and will swap/relocate accordingly.
-- 5) After every swap/move, camera refocuses on your new center.
--
-- Robustness notes:
--  - We never call SetStartingPlot (avoids hardcoded start-distance constraints).
--  - We move ONLY units (the cached "starting group") using remote parking to reduce
--    unintended MEET popups during swap placement.
--  - We track a "center" per player so swaps/undo/manual remain consistent.

local function LOG(msg)
	print(string.format("%s %s | turn=%d active=%d", TAG, tostring(msg), Game.GetGameTurn(), Game.GetActivePlayer()))
end
local function LOGF(fmt, ...) LOG(string.format(fmt, ...)) end
local function SAFE(label, fn)
	local ok, err = pcall(fn)
	if not ok then LOGF("FATAL in %s: %s", tostring(label), tostring(err)) end
	return ok, err
end

	g_GroupUnitIDsByPlayer = {}
	for i = 0, GameDefines.MAX_PLAYERS - 1 do
		local p = Players[i]
		if p and p:IsAlive() then
			local sp = p:GetStartingPlot()

local function IsPlotValidBase(plot)
	if not plot then return false end
	if plot:IsWater() then return false end
	if plot:IsMountain() then return false end
	if plot:IsImpassable() then return false end
	if plot:IsCity() then return false end
	return true
end

	if not IsPlotValidBase(plot) then return false end
	if plot:GetNumUnits() > 0 then return false end
	return true
end
local function FindEmptyPlotsNear(centerPlot, count, startRadius, maxRadius)
	local found = {}
	if not centerPlot then return found end
	local cx, cy = centerPlot:GetX(), centerPlot:GetY()
	for r = startRadius, maxRadius do
		for dx = -r, r do
			for dy = -r, r do
				if #found >= count then return found end
				local p = Map.GetPlot(cx + dx, cy + dy)
				if p and IsPlotPassableAndEmpty(p) then
					table.insert(found, p)
				end
			end
		end
	end
	return found
end
local function FindNearestValidPlot(x, y)
	local base = Map.GetPlot(x, y)
	if base and IsPlotValidBase(base) then
		return base
	end
	for r = MANUAL_NEAREST_SEARCH_START, MANUAL_NEAREST_SEARCH_MAX do
		for dx = -r, r do
			for dy = -r, r do
				local p = Map.GetPlot(x + dx, y + dy)
				if p and IsPlotValidBase(p) then
					return p
				end
			end
		end
	end
	return nil
end
local function FindParkingNearCorner(cornerX, cornerY, count)
	local base = Map.GetPlot(cornerX, cornerY)
	if not base then return {} end
	return FindEmptyPlotsNear(base, count, CORNER_PARK_START_RADIUS, CORNER_PARK_MAX_RADIUS)
end
local function PlaceGroup(units, centerPlot)
	if #units == 0 then return true end
	local spots = FindEmptyPlotsNear(centerPlot, #units, PLACE_RADIUS_START, PLACE_RADIUS_MAX)
	if #spots < #units then return false end
	for i, u in ipairs(units) do
		local p = spots[i]
		u:SetXY(p:GetX(), p:GetY())
	end
	return true
end
local function FindBestParking(count)
	local w, h = Map.GetGridSize()
	local tries = {
		{0, 0},
		{w - 1, h - 1},
		{0, h - 1},
		{w - 1, 0},
	}
	for _, t in ipairs(tries) do
		local spots = FindParkingNearCorner(t[1], t[2], count)
		if #spots >= count then return spots end
	end
	return {}
end
local function ParkUnits(units)
	if #units == 0 then return true end
	local spots = FindBestParking(#units)
	if #spots < #units then return false end
	for i, u in ipairs(units) do
		local p = spots[i]
		u:SetXY(p:GetX(), p:GetY())
	end
	return true
end

-- Core move/swap using centers (NOT starting plot)
------------------------------------------------------------
local function RelocateGroup(playerID, destCenterPlot)
	local groupIDs = g_GroupUnitIDsByPlayer[playerID]
	if not groupIDs then
		LOGF("RelocateGroup: missing cached group for player=%d", playerID)
		return false
	end
	local units = GetUnitsFromIDs(playerID, groupIDs)
	if #units == 0 then
		LOGF("RelocateGroup: empty units for player=%d", playerID)
		return false
	end
	if not destCenterPlot or not IsPlotValidBase(destCenterPlot) then
		return false
	end
	local ok = false
	WithIgnoreUnitSetXY(function()
		if not ParkUnits(units) then ok = false return end
		ok = PlaceGroup(units, destCenterPlot)
	end)
	return ok
end
local function SwapGroupsBetweenCenters(playerA, playerB, centerAPlot, centerBPlot)
	if not centerAPlot or not centerBPlot then return false end
	if not IsPlotValidBase(centerAPlot) then return false end
	if not IsPlotValidBase(centerBPlot) then return false end
	local aIDs = g_GroupUnitIDsByPlayer[playerA]
	local bIDs = g_GroupUnitIDsByPlayer[playerB]
	if not aIDs or not bIDs then
		LOG("SwapGroupsBetweenCenters: missing cached groups")
		return false
	end
	local aUnits = GetUnitsFromIDs(playerA, aIDs)
	local bUnits = GetUnitsFromIDs(playerB, bIDs)
	if #aUnits == 0 or #bUnits == 0 then
		LOG("SwapGroupsBetweenCenters: empty unit list")
		return false
	end
	local ok = false
	WithIgnoreUnitSetXY(function()
		if not ParkUnits(aUnits) then ok = false return end
		if not ParkUnits(bUnits) then ok = false return end
		if not PlaceGroup(aUnits, centerBPlot) then ok = false return end
		if not PlaceGroup(bUnits, centerAPlot) then ok = false return end
		ok = true
	end)
	return ok
end

local function GetOccupyingPlayerAtPlot(plot, excludePlayerID)
	if not plot then return nil end
	local num = plot:GetNumUnits()
	if num <= 0 then return nil end
	for i = 0, num - 1 do
		local u = plot:GetUnit(i)
		if u then
			local owner = u:GetOwner()
			if owner ~= nil and owner ~= excludePlayerID and owner ~= GameDefines.BARBARIAN_PLAYER then
				local p = Players[owner]
				if p and p:IsAlive() then
					return owner
				end
			end
		end
	end
	return nil
end

local function SwapActiveWithTarget(targetID)
	local activeID = Game.GetActivePlayer()

	local centerA = GetCenterPlot(activeID)
	local centerB = GetCenterPlot(targetID)
	if not centerA or not centerB then return false end

	SetCenterPlot(targetID, centerA)

	if not Controls or not Controls.XEdit or not Controls.YEdit then return nil, nil end

local function ManualMoveTo(x, y)
	local activeID = Game.GetActivePlayer()
	if activeID < 0 then return false end
	local w, h = Map.GetGridSize()
	if x == nil or y == nil then
		Notify("TXT_KEY_SRC_BAD_COORDS")
		return false
	end
	if x < 0 or y < 0 or x > (w - 1) or y > (h - 1) then
		Notify("TXT_KEY_SRC_BAD_COORDS")
		return false
	end
	-- If exact plot invalid, choose nearest valid
	local requested = Map.GetPlot(x, y)
	local dest = FindNearestValidPlot(x, y)
	if not dest then
		Notify("TXT_KEY_SRC_BAD_PLOT")
		return false
	end
	local prevCenter = GetCenterPlot(activeID)
	if not prevCenter then
		Notify("TXT_KEY_SRC_MOVE_FAIL")
		return false
	end
	-- If destination occupied by another civ -> swap occupant to prevCenter, us to dest
	local occupantID = GetOccupyingPlayerAtPlot(dest, activeID)
	if occupantID ~= nil then
		LOGF("ManualMoveTo: occupied at (%d,%d) by player=%d; swapping", dest:GetX(), dest:GetY(), occupantID)
		local ok = SwapGroupsBetweenCenters(activeID, occupantID, prevCenter, dest)
		if not ok then
			Notify("TXT_KEY_SRC_MOVE_FAIL")
			return false
		end
		SetCenterPlot(activeID, dest)
		SetCenterPlot(occupantID, prevCenter)
		UI.LookAt(dest)
		if requested ~= dest then
			Notify("TXT_KEY_SRC_MOVE_OK_NEAR")
		else
			Notify("TXT_KEY_SRC_MOVE_OK")
		end
		return true
	end
	-- No occupant: relocate us to dest (even if close to others; we do not enforce start distance rules)
	local ok = RelocateGroup(activeID, dest)
	if not ok then
		Notify("TXT_KEY_SRC_MOVE_FAIL")
		return false
	end
	SetCenterPlot(activeID, dest)
	UI.LookAt(dest)
	if requested ~= dest then
		Notify("TXT_KEY_SRC_MOVE_OK_NEAR")
	else
		Notify("TXT_KEY_SRC_MOVE_OK")
	end
	return true
end

function OnNext()
	if not CanShowUI() then return end
	if #g_TargetPlayerIDs == 0 then
		Notify("TXT_KEY_SRC_NO_TARGETS")
		return
	end
	local targetID = NextTargetID()
	if not targetID then
		Notify("TXT_KEY_SRC_NO_TARGETS")
		return
	end
	if SwapActiveWithTarget(targetID) then
		PushHistory(targetID)
		Notify("TXT_KEY_SRC_SWAP_OK")
	else
		Notify("TXT_KEY_SRC_SWAP_FAIL")
	end
end
function OnPrev()
	if not CanShowUI() then return end
	local last = PopHistory()
	if not last then return end
	if SwapActiveWithTarget(last) then
		Notify("TXT_KEY_SRC_UNDONE")
	else
		Notify("TXT_KEY_SRC_SWAP_FAIL")
	end
end
function OnGoto()
	if not CanShowUI() then return end
	local x, y = ReadManualCoords()

local function OnUnitSetXY(playerID, unitID, x, y)
	-- Any MANUAL unit movement (turn 0) finalizes.
	if g_Finalized then return end

		Finalize("TXT_KEY_SRC_FINALIZED")

	g_Finalized = false
	UpdateUIVisibility()

	if not plot then return nil end
	local num = plot:GetNumUnits()
	if num <= 0 then return nil end
	for i = 0, num - 1 do
		local u = plot:GetUnit(i)
		if u then
			local owner = u:GetOwner()
			if owner ~= nil and owner ~= excludePlayerID and owner ~= GameDefines.BARBARIAN_PLAYER then
				local p = Players[owner]
				if p and p:IsAlive() then

local function ManualMoveTo(x, y)
	local activeID = Game.GetActivePlayer()
	if activeID < 0 then return false end
	local w, h = Map.GetGridSize()
	if x == nil or y == nil then
		Notify("TXT_KEY_SRC_BAD_COORDS")
		return false
	end
	if x < 0 or y < 0 or x > (w - 1) or y > (h - 1) then
		Notify("TXT_KEY_SRC_BAD_COORDS")
		return false
	end
	-- If exact plot invalid, choose nearest valid
	local requested = Map.GetPlot(x, y)
	local dest = FindNearestValidPlot(x, y)
	if not dest then
		Notify("TXT_KEY_SRC_BAD_PLOT")
		return false
	end
	local prevCenter = GetCenterPlot(activeID)
	if not prevCenter then
		Notify("TXT_KEY_SRC_MOVE_FAIL")
		return false
	end
	-- If destination occupied by another civ -> swap occupant to prevCenter, us to dest
	local occupantID = GetOccupyingPlayerAtPlot(dest, activeID)
	if occupantID ~= nil then
		LOGF("ManualMoveTo: occupied at (%d,%d) by player=%d; swapping", dest:GetX(), dest:GetY(), occupantID)
		local ok = SwapGroupsBetweenCenters(activeID, occupantID, prevCenter, dest)
		if not ok then
			Notify("TXT_KEY_SRC_MOVE_FAIL")
			return false
		end
		SetCenterPlot(activeID, dest)
		SetCenterPlot(occupantID, prevCenter)
		UI.LookAt(dest)
		if requested ~= dest then
			Notify("TXT_KEY_SRC_MOVE_OK_NEAR")
		else
			Notify("TXT_KEY_SRC_MOVE_OK")
		end
		return true
	end
	-- No occupant: relocate us to dest (even if close to others; we do not enforce start distance rules)
	local ok = RelocateGroup(activeID, dest)
	if not ok then
		Notify("TXT_KEY_SRC_MOVE_FAIL")
		return false
	end
	SetCenterPlot(activeID, dest)
	UI.LookAt(dest)
	if requested ~= dest then
		Notify("TXT_KEY_SRC_MOVE_OK_NEAR")
	else
		Notify("TXT_KEY_SRC_MOVE_OK")
	end
	return true
end

    if not group then
        return nil

    if group.forms[iEra] then
        return group.forms[iEra]
    end
    if Config.Eras.Future and iEra and iEra >= Config.Eras.Future then

    for _, iPromo in ipairs(Config.AllEraPromotions) do

    if not pUnit or not group then
        return
    end
    local multiplier = Utils.GetEraMultiplier(iEra)
    local newCombat = Utils.Round(group.baseCombat * multiplier)
    pUnit:SetBaseCombatStrength(newCombat)
    if group.baseRangedCombat and pUnit.SetBaseRangedCombatStrength then
        local newRanged = Utils.Round(group.baseRangedCombat * multiplier)
        pUnit:SetBaseRangedCombatStrength(newRanged)
        Utils.Log(string.format(
            "%s synced. Era=%d Step=%d Scale=%.3f Combat=%d Ranged=%d UnitID=%d",
            group.debugName,
            iEra,
            Utils.GetEraStep(iEra),
            multiplier,
            newCombat,
            newRanged,
            pUnit:GetID()
        ))
    else
        Utils.Log(string.format(
            "%s synced. Era=%d Step=%d Scale=%.3f Combat=%d UnitID=%d",
            group.debugName,
            iEra,
            Utils.GetEraStep(iEra),
            multiplier,
            newCombat,
            pUnit:GetID()
        ))
    end

    if not pOldUnit or not pNewUnit then
        return

    local damage = Utils.CallIfExists(pOldUnit, "GetDamage")

        local units = {}
        for pUnit in pPlayer:Units() do
            if not pUnit:IsDead() and GetGroupKeyFromUnitType(pUnit:GetUnitType()) == groupKey then
                table.insert(units, pUnit)
            end
        end
        if #units > 1 then
            table.sort(units, function(a, b)
                return a:GetID() < b:GetID()

        return
    end
    isSyncing = true
    Heroes.EnforceSingleHeroPerGroup(playerID)

    -- iteration in some Civ V Lua runtimes.
    for pUnit in pPlayer:Units() do
        if Heroes.IsHeroUnit(pUnit) then
            table.insert(heroUnits, pUnit)
        end
    end
    for _, pUnit in ipairs(heroUnits) do
        if pUnit and not pUnit:IsDead() then
            SyncHeroUnit(pPlayer, pUnit, iEra)
        end
    end
    isSyncing = false
end

        Heroes.SyncPlayer(iPlayer)
    end
end

        return true
    end

    if not Utils.IsValidPlayer(pPlayer) or not Utils.IsSayajinPlayer(pPlayer) then
        return false
    end
    local groupKey = GetGroupKeyFromUnitType(iUnitType)
    if Heroes.HasLivingHeroInGroup(pPlayer, groupKey) then
        return false
    end
    return true
end

function Utils.Round(value)
    return math.floor(value + 0.5)
end
function Utils.IsValidPlayer(pPlayer)
    return pPlayer
        and pPlayer:IsAlive()
        and not pPlayer:IsBarbarian()
        and not pPlayer:IsMinorCiv()

        and Config

function Utils.GetEraStep(iEra)

        return Config.EraSteps[iEra]
    end
    if Config.Eras.Future and iEra and iEra >= Config.Eras.Future then
        return Config.MaxEraStep
